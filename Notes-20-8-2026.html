#define MAX 100

struct Node {
    int data;
    int next;
};

struct Node list[MAX];

int head = -1;
int avail = 0;

/* Initialize the free-node list */
void initialize()
{
    int i;

    for (i = 0; i < MAX - 1; i++)
        list[i].next = i + 1;

    list[MAX - 1].next = -1;
}

/* Get a free node */
int getNode()
{
    int index;

    if (avail == -1) {
        printf("List is full!\n");
        return -1;
    }

    index = avail;
    avail = list[avail].next;

    return index;
}

/* Return a node to the free list */
void freeNode(int index)
{
    list[index].next = avail;
    avail = index;
}

/* Insert at beginning */
void insertBeginning(int data)
{
    int newNode = getNode();

    if (newNode == -1)
        return;

    list[newNode].data = data;
    list[newNode].next = head;
    head = newNode;
}

/* Insert at end */
void insertEnd(int data)
{
    int newNode, temp;

    newNode = getNode();

    if (newNode == -1)
        return;

    list[newNode].data = data;
    list[newNode].next = -1;

    if (head == -1) {
        head = newNode;
        return;
    }

    temp = head;

    while (list[temp].next != -1)
        temp = list[temp].next;

    list[temp].next = newNode;
}

/* Delete from beginning */
void deleteBeginning()
{
    int temp;

    if (head == -1) {
        printf("List is empty!\n");
        return;
    }

    temp = head;
    head = list[head].next;

    freeNode(temp);
}

/* Display the list */
void display()
{
    int temp = head;

    if (head == -1) {
        printf("List is empty!\n");
        return;
    }

    while (temp != -1) {
        printf("%d -> ", list[temp].data);
        temp = list[temp].next;
    }

    printf("NULL\n");
}

/* Main function */
int main()
{
    int choice, value;

    initialize();

    while (1) {
        printf("\n--- Linked List Using Array ---\n");
        printf("1. Insert at Beginning\n");
        printf("2. Insert at End\n");
        printf("3. Delete from Beginning\n");
        printf("4. Display\n");
        printf("5. Exit\n");

        printf("Enter choice: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                printf("Enter value: ");
                scanf("%d", &value);
                insertBeginning(value);
                break;

            case 2:
                printf("Enter value: ");
                scanf("%d", &value);
                insertEnd(value);
                break;

            case 3:
                deleteBeginning();
                break;

            case 4:
                display();
                break;

            case 5:
                exit(0);

            default:
                printf("Invalid choice!\n");
        }
    }

    return 0;
}
